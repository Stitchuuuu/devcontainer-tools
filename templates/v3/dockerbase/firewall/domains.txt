# =============================================================================
# Firewall allowlist — Claude-only baseline (A1.1, 2026-05-14)
# =============================================================================
# Target: Claude Code itself + its docs + its OFFICIAL Anthropic
# plugins/extensions. Everything else (public npm, GitHub global, MDN,
# packagist, php.net, nodejs.org, DDG, non-npm registries) is opt-in via
# domains.local.txt (see domains.local.txt.example for a commented starter pack).
#
# -----------------------------------------------------------------------------
# Syntax — 5 formats, retro-compatible with A1
# -----------------------------------------------------------------------------
#   Format 1 — bare host (defaults: GET only)
#     docs.anthropic.com
#
#   Format 2 — methods inline (whole host)
#     [GET,POST] api.anthropic.com
#
#   Format 3 — multi-line indented paths (EXACTLY 2 spaces, paths inherit
#              host-level methods; tab/1/3+ spaces = parse error)
#     [*] api.anthropic.com
#       /v1/messages
#       /v1/files
#
#   Format 4 — single-line path + method
#     POST api.anthropic.com/v1/messages
#     [GET,POST] api.anthropic.com/v1/files
#
#   Format 5 — wildcard host and/or path
#     [POST] *.statsig.com                  # wildcard host (all subdomains)
#     [GET] api.github.com/repos/anthropics/*   # trailing * = path wildcard
#
# Local override (domains.local.txt, gitignored):
#     !disable host.com                     # remove a baseline host
#     <any other line>                      # redefine — wipes baseline paths
#                                           # for that host (host-level replace)
# Booted-state warnings: see runtime._overrides_applied in policy.compiled.yaml.
#
# -----------------------------------------------------------------------------
# domains.d/<eco>.txt — project deps per ecosystem (F2)
# -----------------------------------------------------------------------------
# In addition to this file, compile-policy.py walks
# firewall/domains.d/*.txt alphabetically and merges them ADDITIVELY :
# same-host entries across files = methods UNION + paths CONCAT.
#
# Typical layout :
#   domains.d/npm.txt              (generated by extract-auto-dependencies)
#   domains.d/composer.txt         (generated by /scan-deps for composer)
#   domains.d/ecosystem-docs.txt   (curated doc sites for detected libs)
#
# These per-ecosystem files keep this baseline (17 hosts) readable. Project
# dependencies (often 100s of paths) live in domains.d/ where they can be
# regenerated reproducibly.
#
# Same-host entries between domains.txt and domains.d/*.txt merge additively :
#   domains.txt        : [GET] github.com /anthropics/*
#   domains.d/npm.txt  : [GET] github.com /vitejs/vite*
#   → compiled         : [GET] github.com {/anthropics/*, /vitejs/vite*}
#
# -----------------------------------------------------------------------------
# At boot: init-firewall.sh runs compile-policy.py which emits:
#   /var/run/devcontainer-firewall/dnsmasq-domains.conf   (server= + ipset=)
#   /var/run/devcontainer-firewall/policy.compiled.yaml   (consumed by A2 addons)
# Path filters declared below are NOT enforced runtime until A2 ships
# (mitmproxy policy_enforce addon). Currently dnsmasq+ipset = host-level allow.
# =============================================================================

# === CRL / OCSP — TLS certificate revocation infrastructure =================
# TLS chain validation by VS Code / curl / Node / browsers hits these
# endpoints to check whether a presented certificate has been revoked.
# Bodies are signed binary lists (DER-encoded), GET-only, low exfil risk.
# Without these, strict mode generates dozens of host_not_in_policy
# blocks per minute on perfectly normal TLS handshakes.

[GET] crl3.digicert.com                # DigiCert CRL distribution point
[GET] ocsp.digicert.com                # DigiCert OCSP responder
[GET] crl.microsoft.com                # Microsoft CRL distribution point
[GET] ocsp.msocsp.com                  # Microsoft OCSP responder

[GET] www.microsoft.com                # Microsoft PKI Operations — CRL distribution
  /pkiops/crl/*                        #   signed `.crl` binaries (Time-Stamp PCA etc.)


# === Security research — vuln intel + patch level lookup ====================
# Read-only access to public CVE databases (NIST / MITRE / Debian) and
# Docker Desktop release metadata. Purpose : answer « is this CVE real,
# does it affect us, when does the patch land » without leaving the
# devcontainer. GET only, no upload — same risk profile as CRL above.
# Notably useful for kernel CVE where `apt upgrade` is useless (container
# kernel = Docker Desktop's linuxkit, not the FROM image) and the patch
# only ships via Docker Desktop bump.

# --- CVE databases (NIST + MITRE + Debian) ----------------------------------
[GET] nvd.nist.gov                     # NIST NVD — HTML vuln detail pages
  /vuln/*                              #   /vuln/detail/CVE-YYYY-NNNN

[GET] services.nvd.nist.gov            # NVD REST API (JSON, machine-readable)
  /rest/json/cves/*                    #   ?cveId=CVE-YYYY-NNNN

[GET] cve.org                          # MITRE — vanity redirect
[GET] www.cve.org                      # MITRE CVE record pages

[GET] security-tracker.debian.org      # Debian security tracker — patch status per release
  /tracker/*                           #   /tracker/CVE-YYYY-NNNN

[GET] www.debian.org                   # Debian advisories (DSA / DLA)
  /security/*                          #   /security/YYYY/dsa-NNNN

# --- Docker Desktop / linuxkit (kernel patch level for the container host) --
[GET] docs.docker.com                  # Release notes (mentions linuxkit kernel bump)
  /desktop/release-notes/*             #   per-channel release notes

[GET] desktop.docker.com               # Update appcast + installers — latest version metadata (XML)
  /mac/*                               #   /mac/main/{arm64,amd64}/appcast.xml
  /win/*                               #   /win/main/amd64/appcast.xml
  /linux/*                             #   /linux/main/<arch>/ — Docker Desktop Linux release artefacts

# linuxkit project — kernel build recipes + release tags. Docker Desktop's
# bundled VM kernel comes from this repo (kernel/ subtree). Useful to
# cross-reference « ma version Docker Desktop embarque quel kernel patché
# vs vulnérable » quand les release notes Docker n'en parlent pas.
# Merges additively with /repos/anthropics/* and /repos/blunt1337/wtfcmd/*
# declared elsewhere in this file.
[GET] api.github.com
  /repos/linuxkit/linuxkit             #   repo metadata
  /repos/linuxkit/linuxkit/*           #   releases, tags, commits, contents (kernel build files)
